"""
make-cover-from-original.py — 从原文关键图生成公众号封面（用户红线 2026-09：禁止自绘封面）。

用法:
    python make-cover-from-original.py <src_image> <article_dir>

产出:
    <article_dir>/cover.png         900x383（头条封面）
    <article_dir>/cover-square.png  500x500（方图）

规则:
    - 封面 = 挑原文里信息量最大、最能代表文章的一张图（hero/架构图/核心结果图）。
    - 源图 RGBA 先与纯白底 alpha_composite（避免 webp 透明区变黑角）。
    - 等比 contain 居中，不裁主体；背景用同一张图模糊+压暗填满，不写任何文字/标题。
"""
import os
import sys
from PIL import Image, ImageFilter, ImageEnhance


def _load_rgb(path: str) -> Image.Image:
    im = Image.open(path)
    if im.mode in ("RGBA", "LA", "P"):
        im = im.convert("RGBA")
        bg = Image.new("RGBA", im.size, (255, 255, 255, 255))
        im = Image.alpha_composite(bg, im)
    return im.convert("RGB")


def _make(im: Image.Image, size: tuple[int, int]) -> Image.Image:
    W, H = size
    # 背景：cover-fit 模糊 + 压暗（只当底，不是正文）
    bg = im.resize((W, H), Image.LANCZOS).filter(ImageFilter.GaussianBlur(26))
    bg = ImageEnhance.Brightness(bg).enhance(0.55)
    # 前景：等比 contain 居中，绝不裁主体、绝不叠字
    w, h = im.size
    sc = min(W / w, H / h)
    fg = im.resize((int(w * sc), int(h * sc)), Image.LANCZOS)
